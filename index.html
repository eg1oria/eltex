🇷🇺

Переключение языка происходит без перезагрузки страницы — состояние хранится в React useState в HomePage, передаётся во все дочерние компоненты через props.

lib/i18n.ts — createTranslator 
export const LOCALES = ['kk', 'ru', 'en'] as const;
export type Locale = typeof LOCALES[number];

export function createTranslator(locale: Locale) {
  return function t(key: string) {
    return translations[locale][key];
  };
}

// Типизированные данные для каждого компонента
export type AboutItem = {
  icon: string;
  title: string;
  text: string;
};

export type HeroItem = {
  img: string;
  title: string;
  text: string;
};

// Использование в компоненте:
const t = createTranslator(locale);
const items = t('about.items') as AboutItem[];
